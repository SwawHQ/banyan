swban check/browser/csp [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] [--page <path>]  #CSP 执行；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>   在强制 CSP 下加载的页面；默认：/

swban check/browser/csp ./preview
swban check/browser/csp ./preview --headed
swban check/browser/csp ./preview --trace --out ./browser-report
