swban check/browser/prefetch/overlap [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path>  #选择原生机制后不发送自实现预取；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>   含已启用预取候选项的页面

swban check/browser/prefetch/overlap ./preview --page /all/
swban check/browser/prefetch/overlap ./preview --page /discover/topic/ --trace --out ./browser-report
