swban check/browser/language/return [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --from <path> [--page <path>] --to <code>  #语言返回、缺失翻译与状态隔离；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --from <path>   来源页面，保留查询参数和片段
  --page <path>   来源页面链接的语言设置页；默认：/language/
  --to <code>     翻译设置中声明的目标语言

swban check/browser/language/return ./preview --from /article/#from=all&anchor=details --to zh
swban check/browser/language/return ./preview --from /zh/all/#sorts=name-asc --page /zh/language/ --to en --trace
