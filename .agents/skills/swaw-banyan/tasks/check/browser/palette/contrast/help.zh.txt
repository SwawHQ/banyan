swban check/browser/palette/contrast [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path> [--min <ratio>]  #正文与代码对比度；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>   已构建且含正文样式的页面
  --min <ratio>   两种主题的最低文字对比度；默认：4.5

swban check/browser/palette/contrast ./preview --page /p/article/
swban check/browser/palette/contrast ./preview --page /p/article/ --headed
swban check/browser/palette/contrast ./preview --page /p/article/ --trace --out ./browser-report
