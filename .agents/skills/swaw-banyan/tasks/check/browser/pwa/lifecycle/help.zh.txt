swban check/browser/pwa/lifecycle [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --next <dir> [--page <path>] [--from <path>]  #等待更新，关闭受控页面后由浏览器激活；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --next <dir>    下一份启用 SW 的构建，使用不同的 sw.js
  --page <path>   更新面板路径；默认：/powered-by/
  --from <path>   链接到设置页的来源页面；默认：/

swban check/browser/pwa/lifecycle ./old --next ./new
swban check/browser/pwa/lifecycle ./old --next ./new --trace --out ./browser-report
