swban check/browser/pwa/disable [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --next <dir> [--page <path>]  #禁用 SW，清理主题状态并保留其他数据；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --next <dir>    sw.mode = disable 的构建
  --page <path>   清理时访问和重载的页面；默认：/

swban check/browser/pwa/disable ./enabled --next ./disabled
swban check/browser/pwa/disable ./enabled --next ./disabled --page /zh/about/ --trace --out ./browser-report
