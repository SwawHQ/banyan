swban check/browser/nav/sort [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path>  #名称/日期排序与来源导航；Chromium

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>   含不同名称条目和文章链接的集合页

swban check/browser/nav/sort ./preview --page /entries/
swban check/browser/nav/sort ./preview --page /entries/ --headed
swban check/browser/nav/sort ./preview --page /entries/ --trace --out ./browser-report
