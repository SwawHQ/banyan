swban check/browser/aside/layout [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path> [--widths <list>]  #Aside placement, width and wrapping; Chromium

  dir              Published site directory; default: public
  --headed         Show the browser window
  --trace          Save trace.zip; default output: system temporary directory
  --out <dir>      Save JSON and screenshot in a new subdirectory
  --timeout <ms>   Browser launch and individual action timeout; default: 15000
  --page <path>    Published article with an aside
  --widths <list>  Comma-separated viewport widths in pixels; default: 390,1024,1600

swban check/browser/aside/layout ./preview --page /article/
