swban check/browser/aside/layout.webkit [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path> [--widths <list>]  #侧栏位置、宽度与换行；WebKit

  dir              已完成的站点构建目录；默认：public
  --headed         显示浏览器窗口
  --trace          保存 trace.zip；默认输出到系统临时目录
  --out <dir>      在新子目录保存 JSON 和截图
  --timeout <ms>   浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>    已构建且含侧栏的文章
  --widths <list>  视口宽度，单位为像素，用逗号分隔；默认：390,1024,1600

swban check/browser/aside/layout.webkit ./preview --page /article/
