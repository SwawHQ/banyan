swban check/browser/aside/print.webkit [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path>  #打印正文与元信息；WebKit

  dir             已完成的站点构建目录；默认：public
  --headed        显示浏览器窗口
  --trace         保存 trace.zip；默认输出到系统临时目录
  --out <dir>     在新子目录保存 JSON 和截图
  --timeout <ms>  浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>   已构建且含文档元数据的文章

swban check/browser/aside/print.webkit ./preview --page /article/
swban check/browser/aside/print.webkit ./preview --page /article/ --trace --out ./browser-report
