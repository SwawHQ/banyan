swban check/browser/collection/layout [dir] [--headed] [--trace] [--out <dir>] [--timeout <ms>] --page <path> [--article <path>] [--widths <list>]  #名称列宽、表头与行对齐；Chromium

  dir               已完成的站点构建目录；默认：public
  --headed          显示浏览器窗口
  --trace           保存 trace.zip；默认输出到系统临时目录
  --out <dir>       在新子目录保存 JSON 和截图
  --timeout <ms>    浏览器启动和单次操作的超时，单位为毫秒；默认：15000
  --page <path>     含条目和名称列表头的集合页
  --article <path>  来源路径包含该集合的文章；同时比较主列表和面包屑列表
  --widths <list>   视口宽度，用逗号分隔；检查高度为 900 像素；默认：390,1440

swban check/browser/collection/layout ./preview --page /all/
swban check/browser/collection/layout ./preview --page /all/ --article /article/#from=all --trace
