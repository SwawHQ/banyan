{{- $page := . -}} {{- $url := "" -}} {{- $found := false -}} {{- $candidates := slice $page -}} {{- range $page.Ancestors -}} {{- $candidates = $candidates | append . -}} {{- end -}} {{- $candidates = $candidates | append $page.Site.Home -}} {{- range $owner := $candidates -}} {{- if and (not $found) (isset $owner.Params "share_image") -}} {{- $found = true -}} {{- $raw := index $owner.Params "share_image" -}} {{- if eq (printf "%T" $raw) "bool" -}} {{- if $raw -}} {{- errorf "Build failed: page %q has unsupported share_image=true; use a string path/URL or false." ($owner.File.Path | default $owner.RelPermalink) -}} {{- end -}} {{- else -}} {{- $src := trim (printf "%v" $raw) " \r\n\t" -}} {{- if eq $src "" -}} {{- errorf "Build failed: page %q has an empty share_image; use a string path/URL or false." ($owner.File.Path | default $owner.RelPermalink) -}} {{- end -}} {{- if or (strings.HasPrefix $src "http://") (strings.HasPrefix $src "https://") -}} {{- $url = $src -}} {{- else if strings.HasPrefix $src "/" -}} {{- $url = absURL $src -}} {{- else -}} {{- $resolved := partialCached "system-assets/resolve-page.html" (dict "ctx" (dict "Page" $owner) "src" $src) $owner.RelPermalink $src -}} {{- $shareSrc := partialCached "system-metadata/seo/share-image/publish.html" $resolved $owner.RelPermalink $src "share" -}} {{- $url = absURL $shareSrc -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $url -}}
