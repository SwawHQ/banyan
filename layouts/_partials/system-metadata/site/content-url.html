{{- $page := . -}} {{- $url := "" -}} {{- $sourceConfig := $page.Site.Params.content_source | default dict -}} {{- if $sourceConfig -}} {{- if not (reflect.IsMap $sourceConfig) -}} {{- errorf "Build failed: params.content_source must be a map/object." -}} {{- end -}} {{- $repository := trim (printf "%v" (index $sourceConfig "repository" | default "")) " \r\n\t" -}} {{- $branch := trim (printf "%v" (index $sourceConfig "branch" | default "main")) " \r\n\t" -}} {{- $contentRoot := trim (printf "%v" (index $sourceConfig "content_root" | default "content")) " \r\n\t/" -}} {{- $filePath := "" -}} {{- with $page.File -}} {{- $filePath = replace .Path "\\" "/" -}} {{- end -}} {{- if and $repository $branch $contentRoot $filePath -}} {{- $repository = strings.TrimSuffix ".git" (strings.TrimSuffix "/" $repository) -}} {{- $url = printf "%s/blob/%s/%s/%s" $repository $branch $contentRoot (strings.TrimPrefix "/" $filePath) -}} {{- end -}} {{- end -}} {{- return $url -}}
