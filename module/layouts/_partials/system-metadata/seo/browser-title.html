{{- $ctx := . -}} {{- $pageTitle := trim (printf "%v" ($ctx.Params.browser_title | default $ctx.Title | default "")) " \r\n\t" -}} {{- $siteTitle := partial "system-metadata/site/title.html" $ctx.Site -}} {{- $browserTitle := $siteTitle -}} {{- if $ctx.IsHome -}} {{- if ne $pageTitle "" -}} {{- $browserTitle = $pageTitle -}} {{- end -}} {{- else if eq $pageTitle "" -}} {{- $browserTitle = $siteTitle -}} {{- else if eq (lower $pageTitle) (lower $siteTitle) -}} {{- $browserTitle = $pageTitle -}} {{- else -}} {{- $browserTitle = printf "%s - %s" $pageTitle $siteTitle -}} {{- end -}} {{- return $browserTitle -}}
