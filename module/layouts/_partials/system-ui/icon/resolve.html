{{- /* Validate a named SVG, text, or image value. Page resource paths are resolved by system-ui/icon/page-value.html. */ -}} {{- $value := .value -}} {{- $context := .context | default "icon" -}} {{- if eq (printf "%T" $value) "string" -}} {{- $resolved := lower (trim $value " \r\n\t") -}} {{- if and $resolved (not (index hugo.Data.icons $resolved)) -}} {{- errorf "Build failed: %s declares undefined icon %q; define it in data/icons.toml." $context $resolved -}} {{- end -}} {{- return $resolved -}} {{- end -}} {{- if not (reflect.IsMap $value) -}} {{- errorf "Build failed: %s must be an SVG name or an object containing only text, or image with optional monochrome." $context -}} {{- return "" -}} {{- end -}} {{- $isText := and (isset $value "text") (eq (len $value) 1) -}} {{- $isImage := and (isset $value "image") (or (eq (len $value) 1) (and (eq (len $value) 2) (isset $value "monochrome"))) -}} {{- if not (or $isText $isImage) -}} {{- errorf "Build failed: %s must be an SVG name or an object containing only text, or image with optional monochrome." $context -}} {{- return "" -}} {{- end -}} {{- $field := cond $isImage "image" "text" -}} {{- $raw := index $value $field -}} {{- if ne (printf "%T" $raw) "string" -}} {{- errorf "Build failed: %s.%s must be a non-empty string." $context $field -}} {{- return "" -}} {{- end -}} {{- $text := trim $raw " \r\n\t" -}} {{- if not $text -}}{{- errorf "Build failed: %s.%s must be a non-empty string." $context $field -}}{{- end -}} {{- $resolved := dict $field $text -}} {{- if isset $value "monochrome" -}} {{- if ne (printf "%T" $value.monochrome) "bool" -}} {{- errorf "Build failed: %s.monochrome must be a boolean." $context -}} {{- return $resolved -}} {{- end -}} {{- if $value.monochrome -}}{{- $resolved = merge $resolved (dict "monochrome" true) -}}{{- end -}} {{- end -}} {{- return $resolved -}}
