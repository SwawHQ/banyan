{{- $ctx := .ctx | default . -}} {{- /* Slots enable path navigation, article metadata, chapters and BlogPosting. */ -}} {{- $pageID := printf "%s %s" $ctx.Kind $ctx.RelPermalink -}} {{- with $ctx.File -}}{{- $pageID = .Path -}}{{- end -}} {{- $slotConfig := slice -}} {{- if isset $ctx.Params "slots" -}} {{- $raw := index $ctx.Params "slots" -}} {{- if not (reflect.IsSlice $raw) -}} {{- errorf "Build failed: page %q must define slots as a list of slot names." $pageID -}} {{- else -}} {{- $slotConfig = $raw -}} {{- end -}} {{- end -}} {{- $slotFlags := dict "breadcrumb" false "meta" false -}} {{- range $slot := $slotConfig -}} {{- if ne (printf "%T" $slot) "string" -}} {{- errorf "Build failed: page %q only supports string slot names." $pageID -}} {{- else if not (isset $slotFlags $slot) -}} {{- errorf "Build failed: page %q does not support slot %q." $pageID $slot -}} {{- else -}} {{- $slotFlags = merge $slotFlags (dict $slot true) -}} {{- end -}} {{- end -}} {{- return $slotFlags -}}
