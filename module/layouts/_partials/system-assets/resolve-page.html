{{- $ctx := .ctx -}} {{- $src := trim (printf "%v" (.src | default "")) " \r\n\t" -}} {{- $required := true -}} {{- if isset . "required" -}}{{- $required = .required -}}{{- end -}} {{- $result := dict "resolved" false "src" $src -}} {{- $isExternal := or (hasPrefix $src "//") (ne (replaceRE `^[A-Za-z][A-Za-z0-9+.-]*:.*$` "" $src) $src) -}} {{- $isRootRelative := hasPrefix $src "/" -}} {{- if or (eq $src "") $isExternal $isRootRelative -}}{{- return $result -}}{{- end -}} {{- $resourcePath := replaceRE `[#?].*$` "" $src -}} {{- $resourcePath = strings.TrimPrefix "./" $resourcePath -}} {{- $page := $ctx.Page -}} {{- $resource := "" -}} {{- with $page.Resources -}} {{- $resource = .Get $resourcePath -}} {{- end -}} {{- if not $resource -}} {{- if $required -}} {{- errorf "Build failed: page %q cannot resolve local page resource %q. Put the file in the same page bundle, or use an absolute/remote URL." ($page.File.Path | default $page.RelPermalink) $src -}} {{- end -}} {{- return $result -}} {{- end -}} {{- $pathInfo := partial "./page-output-path.html" $page -}} {{- return (dict "resolved" true "resource" $resource "src" $src "targetDir" (index $pathInfo "targetDir")) -}}
