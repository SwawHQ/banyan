{{- $published := .published -}} {{- $resource := index $published "resource" -}} {{- $src := index $published "src" -}} {{- $targetDir := index $published "targetDir" -}} {{- $alt := printf "%v" (.alt | default "") -}} {{- $title := printf "%v" (.title | default "") -}} {{- $stem := partial "system-assets/media/resource-stem.html" $resource -}} {{- $mediaConfig := site.Params.media | default dict -}} {{- $responsiveConfig := index $mediaConfig "responsive_images" | default dict -}} {{- $candidateWidths := index $responsiveConfig "widths" | default (slice 768 1200 1600) -}} {{- $normalizedCandidateWidths := slice -}} {{- range $rawWidth := $candidateWidths -}} {{- $width := int $rawWidth -}} {{- if le $width 0 -}} {{- errorf "Build failed: params.media.responsive_images.widths values must be positive integers, got %q." $rawWidth -}} {{- end -}} {{- $normalizedCandidateWidths = $normalizedCandidateWidths | append $width -}} {{- end -}} {{- $normalizedCandidateWidths = sort (uniq $normalizedCandidateWidths) -}} {{- $variantWidths := slice -}} {{- $targetMaxWidth := int (math.Min $resource.Width (index $normalizedCandidateWidths (sub (len $normalizedCandidateWidths) 1))) -}} {{- range $width := $normalizedCandidateWidths -}} {{- if le $width $targetMaxWidth -}} {{- $variantWidths = $variantWidths | append $width -}} {{- end -}} {{- end -}} {{- if or (eq (len $variantWidths) 0) (ne (index $variantWidths (sub (len $variantWidths) 1)) $targetMaxWidth) -}} {{- $variantWidths = $variantWidths | append $targetMaxWidth -}} {{- end -}} {{- $webpSrcset := slice -}} {{- range $width := $variantWidths -}} {{- $variant := $resource.Resize (printf "%dx webp q78" $width) -}} {{- $targetPath := path.Join $targetDir (printf "%s-%dw.webp" $stem $width) -}} {{- $copied := $variant | resources.Copy $targetPath | fingerprint -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" $copied.RelPermalink $width) -}} {{- end -}} {{/* Fallback estimates support browsers without automatic sizing. */}}
