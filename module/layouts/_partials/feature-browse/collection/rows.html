{{- $page := .page -}} {{- $config := .config -}} {{- if not $config -}}{{- $config = partial "./config.html" (dict "page" $page) -}}{{- end -}} {{- $rows := slice -}} {{- if and (eq $config.provider "aggregate") (not $config.time_tree) -}} {{- $members := partialCached "./members.html" $page $page.Language.Lang $page.Path -}} {{- range $members.direct -}}{{- $rows = $rows | append (partial "feature-browse/collection/rows/page.html" (dict "page" .)) -}}{{- end -}} {{- else -}} {{- $rows = partial "feature-browse/collection/rows/section.html" (dict "page" $page) -}} {{- end -}} {{- $decorated := slice -}} {{- $iconDefaults := partial "system-ui/list/icon-defaults.html" (dict "page" $page) -}} {{- range $row := $rows -}} {{- $node := $row.node -}} {{- if and (eq $config.view "wall") (eq $row.kind "page") -}} {{- $row = merge $row (dict "sort_date" ($node.Date.Format "20060102150405") "date_text" ($node.Date.Format "2006-01-02")) -}} {{- end -}} {{- $icon := partial "system-ui/list/item-icon.html" (dict "page" $node "defaults" $iconDefaults "kind" (cond (eq $row.kind "page") "file" "folder")) -}} {{- $row = merge $row (dict "icon" $icon) -}} {{- if eq $config.view "products" -}} {{- $offer := partial "feature-browse/collection/products/offer.html" (dict "page" $node) -}} {{- $row = merge $row $offer -}} {{- else if eq $config.sort_variant "all" -}} {{- $bytes := len $node.RawContent -}} {{- $row = merge $row (dict "sort_size" (printf "%v" $bytes) "size_text" (printf "%v" (div (add $bytes 1023) 1024)) "size_unit" " KB") -}} {{- end -}} {{- $decorated = $decorated | append $row -}} {{- end -}} {{- /* Hugo owns name collation. Ship its rank so browser locale/ICU cannot reorder the same names during hydration (including mixed scripts). */ -}} {{- $byName := slice -}} {{- range $index, $row := sort $decorated "sort_name" "asc" -}} {{- $byName = $byName | append (merge $row (dict "sort_name" (printf "%d" (add $index 1)))) -}} {{- end -}} {{- $ordered := $byName -}} {{- if and (eq $config.view "wall") (eq $config.provider "section") -}} {{- $byName = collections.Reverse $byName -}} {{- $ordered = slice -}} {{- range sort (where $byName "kind" "ne" "page") "sort_date" "desc" -}}{{- $ordered = $ordered | append . -}}{{- end -}} {{- range sort (where $byName "kind" "page") "sort_date" "desc" -}}{{- $ordered = $ordered | append . -}}{{- end -}} {{- else if eq $config.view "wall" -}} {{- $ordered = sort (collections.Reverse $byName) "sort_date" "desc" -}} {{- else -}} {{- $parts := split $config.default_sort "-" -}} {{- $field := index $parts 0 -}} {{- $order := index $parts 1 -}} {{- if eq $order "desc" -}}{{- $byName = collections.Reverse $byName -}}{{- end -}} {{- $ordered = $byName -}} {{- if ne $field "name" -}} {{- $key := printf "sort_%s" $field -}} {{- /* Numeric keys match browser comparison for dates, counts, sizes and prices. */ -}} {{- $numeric := slice -}} {{- range $row := $byName -}} {{- $numeric = $numeric | append (merge $row (dict "order_value" (float (index $row $key | default "0")))) -}} {{- end -}} {{- $ordered = sort $numeric "order_value" $order -}} {{- if eq $field "price" -}} {{- $ordered = (where $ordered "sort_price" "ne" "") | append (where $ordered "sort_price" "eq" "") -}} {{- end -}} {{- end -}} {{- if eq $config.view "directory" -}} {{- $pages := where $ordered "kind" "eq" "page" -}} {{- $sections := where $ordered "kind" "ne" "page" -}} {{- $ordered = cond (eq $order "asc") ($sections | append $pages) ($pages | append $sections) -}} {{- end -}} {{- end -}} {{- return $ordered -}}
