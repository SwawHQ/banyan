{{/* Compact items payload model; JSON encoding belongs to the publish/embed boundary. */}} {{- $source := .source -}} {{- $config := .config -}} {{- $rows := .rows -}} {{- $fields := slice "kind" "href" "text" "icon" "sort_name" -}} {{- if eq $config.view "products" -}} {{- $fields = $fields | append "sort_price" "sort_value" -}} {{- else if eq $config.view "all" -}} {{- $fields = $fields | append "sort_date" "sort_size" -}} {{- else if eq $config.view "directory" -}} {{- $fields = $fields | append "sort_date" "sort_count" -}} {{- end -}} {{- range $row := $rows -}} {{- if and $row.title (ne $row.title $row.text) -}} {{- $fields = $fields | append "title" -}} {{- break -}} {{- end -}} {{- end -}} {{- $logicalPath := trim (printf "%v" ($source.logicalPath | default "")) " " -}} {{- $rowVector := slice -}} {{- $seenKeys := dict -}} {{- range $row := $rows -}} {{- $key := trim (printf "%v" (index $row "key")) " \r\n\t" -}} {{- $text := trim (printf "%v" (index $row "text")) " \r\n\t" -}} {{- $href := trim (printf "%v" (index $row "href")) " \r\n\t" -}} {{- if or (eq $key "") (eq $text "") (eq $href "") -}} {{- errorf "Build failed: entry source %q has invalid row key=%q text=%q href=%q." $logicalPath $key $text $href -}} {{- end -}} {{- if isset $seenKeys $key -}} {{- errorf "Build failed: entry source %q has duplicate entry key=%q for href=%q and href=%q." $logicalPath $key (index $seenKeys $key) $href -}} {{- end -}} {{- $seenKeys = merge $seenKeys (dict $key $href) -}} {{- range $field := $fields -}} {{- $value := "" -}} {{- if eq $field "title" -}} {{- if ne $row.title $row.text -}}{{- $value = $row.title | default "" -}}{{- end -}} {{- else -}} {{- with index $row $field -}} {{- if eq $field "icon" -}}{{- $value = . -}}{{- else -}}{{- $value = printf "%v" . -}}{{- end -}} {{- end -}} {{- end -}} {{- $rowVector = $rowVector | append $value -}} {{- end -}} {{- end -}} {{- return (dict "v" 1 "p" "collection" "lp" $logicalPath "label" ($source.label | default "") "href" ($source.href | default "") "sv" $config.sort_variant "ds" $config.default_sort "f" $fields "rv" $rowVector ) -}}
