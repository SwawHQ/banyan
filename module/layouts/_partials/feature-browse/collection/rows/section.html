{{- $page := .page -}} {{- if not $page -}} {{- errorf "Build failed: feature-browse/collection/rows/section.html requires page." -}} {{- end -}} {{- if ne $page.Kind "section" -}} {{- errorf "Build failed: feature-browse/collection/rows/section.html only supports kind=section; current page %q has kind=%q." $page.RelPermalink $page.Kind -}} {{- end -}} {{- $directory := partial "feature-browse/directory/members.html" $page -}} {{- $direct := $directory.direct -}} {{- $sectionRows := slice -}} {{- range $section := $page.Sections -}} {{- if not (partial "feature-browse/is-rendered-html.html" .) -}}{{- continue -}}{{- end -}} {{- $membership := partial "feature-browse/directory/members.html" $section -}} {{- $members := $membership.recursive -}} {{- $count := len $members -}} {{- $dateValue := $section.Lastmod -}} {{- with first 1 (sort $members "Lastmod" "desc") -}} {{- $dateValue = (index . 0).Lastmod -}} {{- end -}} {{- $text := $section.LinkTitle | default $section.Title -}} {{- $sectionRows = $sectionRows | append (dict "node" $section "key" (partial "./page-key.html" (dict "ctx" $section)) "kind" "section" "href" $section.RelPermalink "title" $section.Title "text" $text "sort_group" "0" "sort_name" $text "sort_date" (cond $dateValue.IsZero "" ($dateValue.Format "20060102150405")) "date_text" (cond $dateValue.IsZero "—" ($dateValue.Format "2006-01-02")) "sort_count" (printf "%v" $count) "count_text" (printf "%v" $count) "count_unit" "" ) -}} {{- end -}} {{- $pageRows := slice -}} {{- range $child := $direct -}} {{- $pageRows = $pageRows | append (partial "./page.html" (dict "page" $child)) -}} {{- end -}} {{- $rows := slice -}} {{- range sort $pageRows "sort_date" "desc" -}}{{- $rows = $rows | append . -}}{{- end -}} {{- range sort $sectionRows "sort_date" "desc" -}}{{- $rows = $rows | append . -}}{{- end -}} {{- return $rows -}}
