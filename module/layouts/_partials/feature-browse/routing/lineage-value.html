{{- $logicalPath := printf "%v" (.logical_path | default "") -}} {{- $segments := split (trim $logicalPath "/") "/" -}} {{- $cleanSegments := slice -}} {{- range $segment := $segments -}} {{- $value := trim (printf "%v" $segment) " " -}} {{- if ne $value "" -}} {{- $cleanSegments = $cleanSegments | append $value -}} {{- end -}} {{- end -}} {{- /* Match URLSearchParams while keeping path separators readable. */ -}} {{- return (replace (urlquery (delimit $cleanSegments "/")) "%2F" "/") -}}
