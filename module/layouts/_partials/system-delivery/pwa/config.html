{{- $ctx := .ctx | default . -}} {{- $swConfig := .delivery.service_worker -}} {{- $swModeRaw := index $swConfig "mode" | default "off" -}} {{- $swMode := lower (printf "%v" $swModeRaw) -}} {{- if not (in (slice "enable" "disable" "off") $swMode) -}} {{- errorf "Build failed: delivery.service_worker.mode only supports enable/disable/off; got %q." $swModeRaw -}} {{- end -}} {{- $contentCheckInterval := index $swConfig "content_check_interval" -}} {{- if not (and (in (slice "int" "int64") (printf "%T" $contentCheckInterval)) (gt $contentCheckInterval 0)) -}} {{- errorf "Build failed: delivery.service_worker.content_check_interval must be a positive integer in minutes; got %v." $contentCheckInterval -}} {{- end -}} {{- $manifestConfig := $ctx.Site.Params.web_app_manifest | default dict -}} {{- $manifestModeRaw := index $manifestConfig "mode" | default "link" -}} {{- $manifestMode := lower (printf "%v" $manifestModeRaw) -}} {{- if not (in (slice "link" "defer" "off") $manifestMode) -}} {{- errorf "Build failed: params.web_app_manifest.mode only supports link/defer/off; got %q." $manifestModeRaw -}} {{- end -}} {{- $offlinePathRaw := printf "%v" (index $swConfig "offline_notice_path" | default "offline/") -}} {{- $offlinePath := trim $offlinePathRaw "/" -}} {{- if eq $offlinePath "" -}} {{- errorf "Build failed: delivery.service_worker.offline_notice_path cannot be empty." -}} {{- end -}} {{- $offlineRelPermalinks := slice -}} {{- $offlineFallbacks := slice -}} {{- range sort hugo.Sites "Language.Weight" -}} {{- $offlineRelPermalink := printf "%s%s/" .Home.RelPermalink $offlinePath -}} {{- $offlineRelPermalinks = $offlineRelPermalinks | append (lower $offlineRelPermalink) -}} {{- $offlineFallbacks = $offlineFallbacks | append (dict "path_prefix" (printf "%v" .Home.RelPermalink) "url" $offlineRelPermalink ) -}} {{- end -}} {{- return (dict "sw_mode" $swMode "content_check_interval_ms" (mul $contentCheckInterval 60000) "manifest_mode" $manifestMode "offline_fallbacks" $offlineFallbacks "is_offline_page" (in $offlineRelPermalinks (lower $ctx.RelPermalink)) ) -}}
