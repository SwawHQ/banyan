{{- $ctx := .ctx | default . -}} {{- $swConfig := .delivery.sw -}} {{- range $key := slice "mode" "content_check_interval" "offline_notice_path" -}} {{- if not (isset $swConfig $key) -}}{{- errorf "Build failed: delivery.sw requires %s." $key -}}{{- end -}} {{- end -}} {{- $swModeRaw := index $swConfig "mode" -}} {{- $swMode := lower (printf "%v" $swModeRaw) -}} {{- if or (ne (printf "%T" $swModeRaw) "string") (not (in (slice "enable" "disable" "off") $swMode)) -}} {{- errorf "Build failed: delivery.sw.mode only supports enable/disable/off; got %q." $swModeRaw -}} {{- end -}} {{- $contentCheckInterval := index $swConfig "content_check_interval" -}} {{- if not (and (in (slice "int" "int64") (printf "%T" $contentCheckInterval)) (gt $contentCheckInterval 0)) -}} {{- errorf "Build failed: delivery.sw.content_check_interval must be a positive integer in minutes; got %v." $contentCheckInterval -}} {{- end -}} {{- $offlinePathRaw := index $swConfig "offline_notice_path" -}} {{- if ne (printf "%T" $offlinePathRaw) "string" -}}{{- errorf "Build failed: delivery.sw.offline_notice_path must be a string." -}}{{- end -}} {{- $offlinePath := trim $offlinePathRaw "/" -}} {{- if eq $offlinePath "" -}} {{- errorf "Build failed: delivery.sw.offline_notice_path cannot be empty." -}} {{- end -}} {{- $offlineRelPermalinks := slice -}} {{- $offlineFallbacks := slice -}} {{- range sort hugo.Sites "Language.Weight" -}} {{- $offlineRelPermalink := printf "%s%s/" .Home.RelPermalink $offlinePath -}} {{- $offlineRelPermalinks = $offlineRelPermalinks | append (lower $offlineRelPermalink) -}} {{- $offlineFallbacks = $offlineFallbacks | append (dict "path_prefix" (printf "%v" .Home.RelPermalink) "url" $offlineRelPermalink ) -}} {{- end -}} {{- return (dict "sw_mode" $swMode "content_check_interval_ms" (mul $contentCheckInterval 60000) "offline_fallbacks" $offlineFallbacks "is_offline_page" (in $offlineRelPermalinks (lower $ctx.RelPermalink)) ) -}}
