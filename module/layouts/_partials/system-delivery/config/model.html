{{- $policy := dict -}} {{- range $file := slice "policy" "settings" "redirects" -}} {{- $path := printf "deploy/%s.toml" $file -}} {{- with resources.Get $path -}} {{- $values := . | transform.Unmarshal | default dict -}} {{- $allowed := index (dict "settings" (slice "platform" "sw" "prefetch") "policy" (slice "routes" "sw" "prefetch") "redirects" (slice "redirects")) $file -}} {{- range $key, $value := $values -}} {{- if not (in $allowed $key) -}}{{- errorf "Delivery config: unknown field %q in assets/%s." $key $path -}}{{- end -}} {{- if and (in (slice "sw" "prefetch") $key) (reflect.IsMap $value) -}} {{- $switch := cond (eq $key "sw") "mode" "enabled" -}} {{- range $field, $_ := $value -}} {{- if ne (eq $file "settings") (eq $field $switch) -}}{{- errorf "Delivery config: %s.%s belongs in assets/deploy/%s.toml." $key $field (cond (eq $field $switch) "settings" "policy") -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $policy = merge $policy $values -}} {{- else -}} {{- if ne $file "redirects" -}} {{- errorf "Build failed: missing assets/%s." $path -}} {{- end -}} {{- end -}} {{- end -}} {{- partial "./validate.html" $policy -}} {{- $routes := index $policy "routes" | default slice -}} {{- $redirects := partial "system-delivery/deployment/redirect-rules.html" (dict "rules" (index $policy "redirects" | default slice)) -}} {{- $swRoutes := slice -}} {{- range $route := $routes -}} {{- with index $route "sw" -}} {{- $swRoutes = $swRoutes | append (merge . (dict "path" (index $route "path"))) -}} {{- end -}} {{- end -}} {{- /* Source exceptions come last; the worker returns its first matching rule. */ -}} {{- return (merge $policy (dict "routes" $routes "redirects" $redirects "sw_routes" (collections.Reverse $swRoutes))) -}}
