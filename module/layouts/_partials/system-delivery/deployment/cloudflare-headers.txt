{{- $routes := partial "system-delivery/deployment/cache-header-routes.html" (dict "routes" (.routes | default slice) "platform" "cloudflare" "product" "pages") -}}
{{- $lines := slice -}}

{{- $lines = $lines | append "# GENERATED FILE: edit data/delivery.toml or Banyan's deployment templates." -}}
{{- $lines = $lines | append "" -}}

{{- range $route := $routes -}}
    {{- $path := printf "%v" (index $route "path" | default "") -}}
    {{- if not $path -}}
        {{- continue -}}
    {{- end -}}
    {{- $lines = $lines | append $path -}}
    {{- range .unset -}}
        {{- $lines = $lines | append (printf "  ! %s" .) -}}
    {{- end -}}
    {{- $headers := index $route "headers" | default dict -}}
    {{- range $key, $value := $headers -}}
        {{- if strings.HasPrefix $key "#" -}}
            {{- errorf "Cloudflare _headers: header %q for %q must not start with #; it would be treated as a comment." $key $path -}}
            {{- continue -}}
        {{- end -}}
        {{- $lines = $lines | append (printf "  %s: %s" $key $value) -}}
    {{- end -}}
    {{- $lines = $lines | append "" -}}
{{- end -}}

{{- $body := printf "%s\n" (delimit $lines "\n") -}}
{{- return $body -}}
