{{- $routes := partial "./cache-header-routes.html" (dict "routes" (.routes | default slice) "platform" "edgeone" "product" "makers") -}} {{- $redirectRules := .redirects | default slice -}} {{- $redirects := slice -}} {{- $headers := slice -}} {{- range $rule := $redirectRules -}} {{- $redirects = $redirects | append (dict "source" (index $rule "source") "destination" (index $rule "destination") "statusCode" (index $rule "status") ) -}} {{- end -}} {{- range $route := $routes -}} {{- $path := printf "%v" (index $route "path" | default "") -}} {{- if not $path -}} {{- continue -}} {{- end -}} {{- $headerEntries := slice -}} {{- range $key, $value := (index $route "headers" | default dict) -}} {{- $headerEntries = $headerEntries | append (dict "key" $key "value" $value) -}} {{- end -}} {{- $headers = $headers | append (dict "source" $path "headers" $headerEntries ) -}} {{- end -}} {{- $config := dict "redirects" $redirects "headers" $headers -}} {{- return $config -}}
