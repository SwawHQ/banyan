{{- $rules := .rules | default slice -}} {{- $normalized := slice -}} {{- range $i, $rule := $rules -}} {{- if not (reflect.IsMap $rule) -}} {{- errorf "Delivery config: redirects[%d] must be a table." $i -}} {{- continue -}} {{- end -}} {{- $source := printf "%v" (index $rule "source" | default "") -}} {{- $destination := printf "%v" (index $rule "destination" | default "") -}} {{- $status := int (index $rule "status" | default 0) -}} {{- if or (eq $source "") (eq $destination "") (eq $status 0) -}} {{- errorf "Build failed: assets/site/delivery.toml redirects require non-empty source, destination, and status values." -}} {{- end -}} {{- if not (in (slice 301 302) $status) -}} {{- errorf "Build failed: assets/site/delivery.toml only supports status 301 or 302 for redirects; got %d." $status -}} {{- end -}} {{- $normalized = $normalized | append (dict "source" $source "destination" $destination "status" $status ) -}} {{- end -}} {{- return $normalized -}}
