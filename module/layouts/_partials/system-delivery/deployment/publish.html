{{- $routes := .routes | default slice -}} {{- $redirects := .redirects | default slice -}} {{- $platform := .delivery.platform | default dict -}} {{- $headerRoutes := $routes -}} {{- $prefetch := partialCached "system-delivery/prefetch/config.html" (dict "delivery" .delivery) "prefetch" -}} {{- $nativeBody := partial "system-delivery/prefetch/native.html" (dict "config" $prefetch) -}} {{- $csp := partial "./csp-value.html" (merge .inline_scripts (dict "prefetch_native" $nativeBody)) -}} {{- $updatedRoutes := slice -}} {{- $hasGlobal := false -}} {{- $generatedHeaders := dict $updatedRoutes = $updatedRoutes | append $route -}} {{- end -}} {{- $headerRoutes = $updatedRoutes -}} {{- if not $hasGlobal -}} {{- $headerRoutes = (slice (dict "path" "/*" "http" (dict "headers" $generatedHeaders))) | append $headerRoutes -}} {{- end -}} {{- if eq $platform.cloudflare.pages.enabled true -}} {{- $headersBody := partial "./cloudflare-headers.txt" (dict "routes" $headerRoutes) -}} {{- $publishHeaders := resources.FromString "_headers" $headersBody | resources.Publish -}} {{- $redirectsBody := partial "./cloudflare-redirects.txt" (dict "redirects" $redirects) -}} {{- $publishRedirects := resources.FromString "_redirects" $redirectsBody | resources.Publish -}} {{- end -}} {{- if eq $platform.edgeone.makers.enabled true -}} {{- $edgeoneConfig := partial "./edgeone-config.html" (dict "edgeone" $platform.edgeone.makers "routes" $headerRoutes "redirects" $redirects ) -}} {{- $publishEdgeone := resources.FromString "edgeone.json" ($edgeoneConfig | jsonify (dict "indent" " ")) | resources.Publish -}} {{- end -}}
