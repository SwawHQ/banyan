{{- $config := .config -}} {{- $context := .context -}} {{- $values := dict "default" "off" -}} {{- if isset $config "default" -}}{{- $values = dict "default" $config.default -}}{{- end -}} {{- $icons := dict -}} {{- if isset $config "icons" -}}{{- $icons = $config.icons -}}{{- end -}} {{- if not (reflect.IsMap $icons) -}} {{- errorf "Build failed: %s.icons must be a table." $context -}} {{- $icons = dict -}} {{- end -}} {{- $library := partialCached "system-ui/icon/library.html" site "site-icons" -}} {{- range $icon, $scopes := $icons -}} {{- if or (eq (len (findRE "^[a-z][a-z0-9_-]*$" $icon)) 0) (not (index $library $icon)) -}} {{- errorf "Build failed: %s.icons.%s must name an SVG defined in assets/site/icons.toml." $context $icon -}} {{- end -}} {{- if not (reflect.IsMap $scopes) -}} {{- errorf "Build failed: %s.icons.%s must be a scope table." $context $icon -}} {{- else -}} {{- range $scope, $value := $scopes -}} {{- if not (or (in (slice "all" "root" "tail") $scope) (gt (len (findRE "^(root\\+|tail-)[1-9][0-9]*$" $scope)) 0)) -}} {{- errorf "Build failed: %s.icons.%s has unknown scope %q; use all/root/tail/root+/tail-." $context $icon $scope -}} {{- end -}} {{- $values = merge $values (dict (printf "%s/%s" $icon $scope) $value) -}} {{- end -}} {{- end -}} {{- end -}} {{- $normalized := dict -}} {{- range $id, $value := $values -}} {{- $mode := lower (trim (printf "%v" $value) " \r\n\t") -}} {{- if or (ne (printf "%T" $value) "string") (not (in (slice "off" "conservative" "moderate" "eager" "immediate") $mode)) -}} {{- errorf "Build failed: %s rule %s must use off/conservative/moderate/eager/immediate; got %q." $context $id $value -}} {{- end -}} {{- $normalized = merge $normalized (dict $id $mode) -}} {{- end -}} {{- return $normalized -}}
